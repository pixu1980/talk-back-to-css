Button API from HTML

data-tone, data-size, and data-radius become the component contract. 

        <button is="back-button" data-tone="brand" data-size="3" data-radius="12">
          Save changes
        </button>

        <button is="back-button" data-tone="danger" data-size="3" data-radius="999">
          Delete project
        </button>
      

        [is='back-button'] {
          --tone: attr(data-tone type(<custom-ident>), brand);
          --size: attr(data-size type(<number>), 3);
          --radius: attr(data-radius px, 12px);

          --accent: if(style(--tone: danger): #d92d20; style(--tone: ghost): transparent; else: #1252ff;);
          --text: if(style(--tone: ghost): #1252ff; else: white;);

          padding: calc(var(--size) * 0.25rem) calc(var(--size) * 0.5rem);
          border: 1px solid if(style(--tone: ghost): #1252ff; else: transparent;);
          border-radius: var(--radius);
          background: var(--accent);
          color: var(--text);
        }
