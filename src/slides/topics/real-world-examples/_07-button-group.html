Button Group

A design system edge case solved by attribute contracts and conditional values. 

          <div is="back-button-group" data-axis="inline">
            <button data-position="first">Left</button>
            <button data-position="middle">Center</button>
            <button data-position="last">Right</button>
          </div>
        

          [is='back-button-group'] {
            --axis: attr(data-axis type(<custom-ident>), inline);

            display: flex;
            flex-direction: if(style(--axis: block): column; else: row);
          }

          [is='back-button-group'] > button {
            --position: attr(data-position type(<custom-ident>), single);
            --radius: 0.75rem;

            border: 1px solid #1252ff;
            padding: 0.75rem 1rem;
            border-radius: if(
              style((--position: first) and (--axis: inline)): var(--radius) 0 0 var(--radius);
              style((--position: last) and (--axis: inline)): 0 var(--radius) var(--radius) 0;
              style((--position: first) and (--axis: block)): var(--radius) var(--radius) 0 0;
              style((--position: last) and (--axis: block)): 0 0 var(--radius) var(--radius);
              else: 0;
            );
            margin-inline-start: if(style((--axis: inline) and ((--position: middle) or (--position: last))): -1px; else: 0;);
            margin-block-start: if(style((--axis: block) and ((--position: middle) or (--position: last))): -1px; else: 0;);
          }
